TanStack Query v5 optimistic update pattern:
```typescript
const mutation = useMutation({
  mutationFn: ({ id, data }: { id: number; data: PatchTrendPayload }) => patchTrend(id, data),
  onMutate: async ({ id, data }) => {
    await queryClient.cancelQueries({ queryKey: ['trends'] });
    const previous = queryClient.getQueryData<PaginatedTrends>(['trends', ...]);
    // update cache optimistically
    queryClient.setQueryData(['trends', ...], (old: PaginatedTrends | undefined) => {
      if (!old) return old;
      return { ...old, items: old.items.map(t => t.Id === id ? { ...t, ...data } : t) };
    });
    return { previous };
  },
  onError: (_err, _vars, context) => {
    if (context?.previous) queryClient.setQueryData(['trends', ...], context.previous);
    toast.error('Action failed');
  },
  onSuccess: () => toast.success('Updated'),
});
```

**Add patch mutation (for approve/reject):**
```typescript
const patchMutation = useMutation({
  mutationFn: ({ id, data }: { id: number; data: PatchTrendPayload }) =>
    patchTrend(id, data),
  onMutate: async ({ id, data }) => {
    // Cancel any in-flight refetch so it doesn't overwrite optimistic update
    await queryClient.cancelQueries({ queryKey: ['trends'] });
    // Save current cache for rollback
    const previousData = queryClient.getQueryData<PaginatedTrends>(
      ['trends', filters, cursor, sortBy, sortDir]
    );
    // Apply optimistic update
    queryClient.setQueryData<PaginatedTrends>(
      ['trends', filters, cursor, sortBy, sortDir],
      (old) => {
        if (!old) return old;
        return {
          ...old,
          items: old.items.map((t) => (t.Id === id ? { ...t, ...data } : t)),
        };
      }
    );
    return { previousData };
  },
  onError: (_err, _vars, context) => {
    if (context?.previousData) {
      queryClient.setQueryData(
        ['trends', filters, cursor, sortBy, sortDir],
        context.previousData
      );
    }
    toast.error('Action failed');
  },
  onSuccess: (_data, { data }) => {
    toast.success(
      data.status === 'approved' ? 'Approved' :
      data.status === 'rejected' ? 'Rejected' : 'Updated'
    );
  },
});
```
