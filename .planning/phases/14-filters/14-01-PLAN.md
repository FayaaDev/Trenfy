Create `WhiteLabelApp/src/hooks/useFilterPrefs.ts`.

Use `Storage` from `expo-sqlite/kv-store` and keep the hook synchronous so platform and region are available before the first feed request.

Use these exact keys:
```typescript
const PLATFORM_KEY = 'filter:platform';
const REGION_KEY = 'filter:region';
```

Implementation shape:
```typescript
import { useCallback, useState } from 'react';
import Storage from 'expo-sqlite/kv-store';
import type { TrendPlatform, TrendRegion } from '../types';

function readPlatform(): TrendPlatform | null { ... }
function readRegion(): TrendRegion | null { ... }

export interface UseFilterPrefsResult {
  platform: TrendPlatform | null;
  region: TrendRegion | null;
  setPlatform: (platform: TrendPlatform | null) => void;
  setRegion: (region: TrendRegion | null) => void;
  clearPersistedFilters: () => void;
}

export default function useFilterPrefs(): UseFilterPrefsResult {
  const [platform, setPlatformState] = useState<TrendPlatform | null>(() => readPlatform());
  const [region, setRegionState] = useState<TrendRegion | null>(() => readRegion());

  const setPlatform = useCallback((next: TrendPlatform | null) => {
    setPlatformState(next);
    if (next) Storage.setItemSync(PLATFORM_KEY, next);
    else Storage.removeItemSync(PLATFORM_KEY);
  }, []);

  const setRegion = useCallback((next: TrendRegion | null) => {
    setRegionState(next);
    if (next) Storage.setItemSync(REGION_KEY, next);
    else Storage.removeItemSync(REGION_KEY);
  }, []);

  const clearPersistedFilters = useCallback(() => {
    setPlatformState(null);
    setRegionState(null);
    Storage.removeItemSync(PLATFORM_KEY);
    Storage.removeItemSync(REGION_KEY);
  }, []);

  return { platform, region, setPlatform, setRegion, clearPersistedFilters };
}
```

Validation details:
- Guard against unexpected stored values by accepting only `youtube`, `x`, `US`, `SA`, and `JP`
- Treat empty strings and unknown values as `null`
- No async hydration effect; use lazy `useState` initializers only

Key behavior changes:
- Keep `searchQuery`, `refresh`, `loadMore`, loading flags, and error state exactly as the public API
- Maintain a private `rawItems` array for the server response
- Fetch from the API using only server-supported params: `platform`, `region_code`, `q`, `cursor`, `limit`
- Do not send `category` when multiple categories are active; apply category narrowing in a `useMemo` over `rawItems`
- Reset cursor and reload page 1 when `filters.platform`, `filters.regionCode`, or `searchQuery` changes
- Category chip changes should update visible items immediately without a new network request
- Deduplicate appended items by `id` during pagination

Recommended structure:
```typescript
export default function useTrendFeed(filters: TrendFeedFilters): UseTrendFeedResult {
  const [rawItems, setRawItems] = useState<Trend[]>([]);
  // keep current loading/search state

  const selectedCategories = filters.selectedCategories;

  const items = useMemo(() => {
    if (selectedCategories.length === 0) return rawItems;
    return rawItems.filter((item) =>
      item.category ? selectedCategories.includes(item.category) : false
    );
  }, [rawItems, selectedCategories]);
}
```

Recommended server fetch call:
```typescript
const result = await fetchTrends({
  platform: filters.platform ?? undefined,
  region_code: filters.regionCode ?? undefined,
  q: opts.q || undefined,
  cursor: opts.cursor || undefined,
  limit: 20,
});
```

Dependency guidance:
- Create a stable `serverFilterKey = `${filters.platform ?? ''}|${filters.regionCode ?? ''}``
- Use that key in the reload effect so platform/region changes re-fetch cleanly
- Keep the existing mounted ref to avoid double-fetching on mount for debounced search

Do not add category fetching or persistence logic here; that stays outside this hook.
